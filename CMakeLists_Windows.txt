cmake_minimum_required(VERSION 3.10)
project(kcobain C CXX)

# Windows-specific configuration for GCC 6.3.0
if(WIN32)
    # Force GCC compiler detection
    set(CMAKE_C_COMPILER gcc)
    set(CMAKE_CXX_COMPILER g++)
    
    # Set C++ standard to C++11 (compatible with GCC 6.3.0)
    set(CMAKE_CXX_STANDARD 11)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    
    # Windows-specific compiler flags for GCC 6.3.0
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++11 -Wall -Wextra -O2")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -O2")
    
    # Windows-specific definitions
    add_definitions(-DWIN32)
    add_definitions(-D_WIN32)
    add_definitions(-D_WINDOWS)
    
    # Disable Windows-specific warnings that might be problematic with GCC
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-unused-parameter")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-unused-variable")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-unused-function")
    
    # Enable position independent code for better compatibility
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fPIC")
    
    # Set Windows subsystem (console application)
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -mconsole")
    
    # Disable specific warnings that might appear with GCC on Windows
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-format")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-sign-compare")
    
    # Enable debug symbols in debug builds
    set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -g -O0")
    set(CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG} -g -O0")
    
    # Optimize release builds
    set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -DNDEBUG")
    set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -O3 -DNDEBUG")
    
    message(STATUS "Windows GCC build configuration detected")
    message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
    message(STATUS "C++ Flags: ${CMAKE_CXX_FLAGS}")
else()
    message(FATAL_ERROR "This CMakeLists.txt is for Windows builds only. Use CMakeLists.txt for other platforms.")
endif()

# Project configuration
set(LOGGER_TAG "${PROJECT_NAME}" CACHE STRING "Logger tag for the application")

# Add include directories
include_directories(${CMAKE_SOURCE_DIR}/include)
include_directories(${CMAKE_SOURCE_DIR}/external)

# Define project constants for the application
add_definitions(-DLOGGER_TAG="${LOGGER_TAG}")
add_definitions(-DPROJECT_NAME="${PROJECT_NAME}")
add_definitions(-DKCobain_VERSION="1.0.0")

# Copy header files to build directory (for development)
file(COPY ${CMAKE_SOURCE_DIR}/include DESTINATION ${CMAKE_BINARY_DIR})

# Create core audio library (audio_rb_controller only)
add_library(kcobain_core STATIC
    src/utils/logger.cpp
    src/miniaudio_impl.cpp
    src/core/audio_rb_controller.cpp
)

# Create USB audio library (separate from core)
add_library(kcobain_usb STATIC
    src/core/usb_audio_producer.cpp
    src/core/usb_audio_consumer.cpp
    src/core/usb_audio_orchestrator.cpp
)

# Link core library to USB library
target_link_libraries(kcobain_usb kcobain_core)

# Create main executable (depends on USB library)
add_executable(kcobain 
    src/main.cpp
)

# Link libraries to main executable
target_link_libraries(kcobain kcobain_usb)

# Windows-specific library linking
if(WIN32)
    # Link Windows system libraries
    target_link_libraries(kcobain_core kernel32 user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
    target_link_libraries(kcobain_usb kernel32 user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
    target_link_libraries(kcobain kernel32 user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
    
    # Add math library for Windows GCC
    target_link_libraries(kcobain m)
    
    # Set Windows-specific properties
    set_target_properties(kcobain PROPERTIES
        WIN32_EXECUTABLE FALSE  # Console application
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    )
    
    # Set library output directories
    set_target_properties(kcobain_core PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
    )
    
    set_target_properties(kcobain_usb PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
    )
endif()

# Install headers (for library usage)
install(DIRECTORY ${CMAKE_SOURCE_DIR}/include/kcobain
        DESTINATION include
        FILES_MATCHING PATTERN "*.h")

# Install libraries
install(TARGETS kcobain_core kcobain_usb
        ARCHIVE DESTINATION lib
        LIBRARY DESTINATION lib)

# Install executable
install(TARGETS kcobain
        RUNTIME DESTINATION bin)

# Print configuration summary
message(STATUS "=== KCobain Windows Build Configuration ===")
message(STATUS "Project: ${PROJECT_NAME}")
message(STATUS "Version: ${KCobain_VERSION}")
message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "Output Directory: ${CMAKE_BINARY_DIR}/bin")
message(STATUS "==========================================")
